function getFilteredSessions(source = db) {
  const classroom = (session) =>
    source.schools
      .find((school) => school.id === session.schoolId)
      ?.classes.find((item) => item.id === session.classId);
  return source.sessions
    .filter((s) => {
      if (globalFilter.date && s.date !== globalFilter.date) return false;
      if (globalFilter.schoolId && s.schoolId !== globalFilter.schoolId)
        return false;
      if (globalFilter.classId && s.classId !== globalFilter.classId)
        return false;
      if (globalFilter.lessonId && s.lessonId !== globalFilter.lessonId)
        return false;
      if (globalFilter.period && String(s.period) !== globalFilter.period)
        return false;
      if (globalFilter.studentId) {
        const hasAtt = s.attendance[globalFilter.studentId] !== undefined;
        const hasScore = s.scores.some(
          (x) => x.studentId === globalFilter.studentId,
        );
        const roster = s.roster || classroom(s)?.students || [];
        if (
          !hasAtt &&
          !hasScore &&
          !roster.some((st) => st.id === globalFilter.studentId)
        )
          return false;
      }
      return true;
    })
    .map((session) => {
      const studentId = globalFilter.studentId;
      if (!studentId) return session;
      // Scope the complete session projection, not just the list of sessions.
      // The detailed renderer and copied text must not expose other students.
      const roster = session.roster || classroom(session)?.students || [];
      return {
        ...session,
        roster: roster.filter((student) => student.id === studentId),
        attendance: Object.hasOwn(session.attendance, studentId)
          ? { [studentId]: session.attendance[studentId] }
          : {},
        scores: session.scores.filter((score) => score.studentId === studentId),
      };
    })
    .sort((a, b) => {
      if (a.date !== b.date) return b.date.localeCompare(a.date);
      return a.period - b.period;
    });
}

function activityReportAttendance(classroom, session) {
  const counts = EduReportAttendance.emptyCounts();
  const people = sessionStudents(classroom, session).map((student) => {
    const attendance = EduReportAttendance.normalize(
      session.attendance[student.id],
    );
    EduReportAttendance.tally(counts, attendance);
    return { student, attendance };
  });
  return { counts, people };
}

      h += `<th>${header}</th>`;
    });
    h += `</tr></thead><tbody>`;

            sess.attendance[globalFilter.studentId],

            attText = '<span class="badge absent">غایب</span>';
          else if (st.status === "late") {
            attText = `<span class="badge late">تأخیر${st.minutes === null ? "" : " (" + st.minutes + "د)"}</span>`;
          }
          h += `<td>${attText}</td>`;
        }
        const myScores = sess.scores.filter(
          (x) => x.studentId === globalFilter.studentId,
        );
        if (globalFilter.showScores) {
          const ms = myScores
            .filter((x) => x.type === "mostamar")
            .map((x) => `${x.value}`)
            .map(esc)
            .join("، ");
          const ps = myScores
            .filter((x) => x.type === "payani")
            .map((x) => `${x.value}`)
            .map(esc)
            .join("، ");

            .filter((score) => score.type === "classGrade")
            .map((score) => score.value)
            .join("، ");
          const activityMarks = myScores
            .filter((score) => score.type === "activity")
            .map((score) => score.value)
            .join("، ");

            "<td>" +
            (ms || "—") +
            " / " +
            (ps || "—") +
            (classMarks ? "<br>کلاسی: " + esc(classMarks) : "") +
            (activityMarks ? "<br>فعالیت: " + esc(activityMarks) : "") +
            "</td>";

          const pos = myScores.filter((x) => x.type === "positive");

                  .map((x) => `(${x.value})`)

                  .join("، ")

          }</td>`;

          const hw = myScores.filter((x) => x.type === "homework");
          h += `<td>${hw.length ? hw.length : '<span class="muted">0</span>'}</td>`;
        }
      } else {
        if (globalFilter.showAttendance) {
          const { counts } = activityReportAttendance(cls.classObj, sess);
          h += `<td>✅${counts.present + counts.late} ❌${counts.absent} ⏰${counts.late} · مرخصی ${counts.leave} · تعیین‌نشده ${counts.unknown}</td>`;
        }
        if (globalFilter.showScores) {
          const m = sess.scores.filter((x) => x.type === "mostamar").length;
          const pay = sess.scores.filter((x) => x.type === "payani").length;
          const classMarks = sess.scores.filter(
            (score) => score.type === "classGrade",

            (score) => score.type === "activity",
          ).length;

                m +
                " · پایانی: " +
                pay +

        if (globalFilter.showPosNeg) {

          const neg = sess.scores.filter((x) => x.type === "negative").length;

          const presentNames = [],
            absentNames = [],
            leaveNames = [],
            lateNames = [],
            unknownNames = [];
          activityReportAttendance(cls.classObj, sess).people.forEach(

              if (attendance.status === "present") presentNames.push(st.name);
              else if (attendance.status === "leave") leaveNames.push(st.name);
              else if (attendance.status === "absent")

              else if (attendance.status === "late") {
                presentNames.push(st.name);
                lateNames.push(
                  `${st.name}${attendance.minutes === null ? "" : " (" + attendance.minutes + "د)"}`,
                );
              } else unknownNames.push(st.name);

            h += `<div class="names-list"><b class="present">✅ حاضر:</b> ${presentNames.length} نفر</div>`;

          const pos = sess.scores.filter((x) => x.type === "positive");

          if (pos.length)
            h +=
              `<div class="names-list" style="color:var(--status-success, #16664e);margin-top:4px">➕ مثبت (${pos.length}): ` +
              pos
                .map((x) => {
                  const st = sessionStudent(cls.classObj, sess, x.studentId);
                  return `${st ? st.name : ""}(${x.value})`;
                })
                .map(esc)
                .join("، ") +
              `</div>`;
          if (neg.length)
            h +=
              `<div class="names-list" style="color:var(--status-danger, #9f2635);margin-top:4px">➖ منفی (${neg.length}): ` +
              neg
                .map((x) => {
                  const st = sessionStudent(cls.classObj, sess, x.studentId);
                  return `${st ? st.name : ""}(${x.value})`;
                })
                .map(esc)
                .join("، ") +
              `</div>`;
        }
        if (globalFilter.showHomework) {
          const hw = sess.scores.filter((x) => x.type === "homework");
          if (hw.length)
            h +=
              `<div class="names-list" style="color:var(--status-info, #1d4f91);margin-top:4px">📝 تکالیف (${hw.length}): ` +
              hw
                .map((x) => {
                  const st = sessionStudent(cls.classObj, sess, x.studentId);
                  return `${st ? st.name : ""}${x.note ? "(" + x.note + ")" : ""}`;
                })
                .map(esc)
                .join("، ") +
              `</div>`;
        }
        h += `</div><hr style="border:0;border-top:1px dashed var(--theme-line, #e4e4e7);margin:10px 0">`;
      });
      h += `</div>`;
    });
    h += `</div>`;
  }
  h += `</div>`;
  out.innerHTML = h;
}

function copyGlobalReportText(btn) {
  return EduV24.copyReport(buildGlobalReportText, btn);
}

function buildGlobalReportText(source = db) {

    for (const school of source.schools) {
      if (schoolId && school.id !== schoolId) continue;
      const classObj = school.classes.find((item) => item.id === classId);
      if (classObj) return { school, classObj };
    }
    return null;
  };
  const filteredSessions = getFilteredSessions(source);
  if (filteredSessions.length === 0) return alert("موردی برای گزارش یافت نشد.");
  let text = `📊 گزارش جامع عملکرد کلاسی\n`;
  if (globalFilter.date)
    text += `📅 تاریخ گزارش: ${faDate(globalFilter.date)}\n`;
  else text += `📅 تاریخ تهیه گزارش: ${faDate(todayISO())}\n`;
  if (globalFilter.classId) {
    const cls = findClass(globalFilter.classId, globalFilter.schoolId);
    text += `🏫 مدرسه: ${cls.school.name} | کلاس: ${cls.classObj.name}\n`;
  }
  text += `================================\n\n`;
  if (globalFilter.reportMode === "summary") {
    filteredSessions.forEach((sess) => {
      const cls = findClass(sess.classId, sess.schoolId);
      const lesson = getLessonObj(cls.classObj, sess.lessonId);
      text += `📅 ${faDate(sess.date)} | کلاس: ${cls.classObj.name} | درس: ${lesson.name} | ${getPeriodName(sess.period)}\n`;
      if (globalFilter.studentId) {
        const stObj = sessionStudent(
          cls.classObj,

            else text += `   ❌ غایبین: ${absentNames.length} نفر\n`;
          }
          if (lateNames.length > 0) {
            if (globalFilter.showLateNames)
              text += `   ⏰ تأخیری‌ها (${lateNames.length}): ${lateNames.join("، ")}\n`;
            else text += `   ⏰ تأخیری‌ها: ${lateNames.length} نفر\n`;
          }
        }
        if (globalFilter.showScores) {
          const ex = sess.scores.filter(
            (x) =>
              x.type === "mostamar" ||
              x.type === "payani" ||
              x.type === "classGrade" ||
              x.type === "activity",
          );
          if (ex.length)
            text +=
              `   📊 نمرات: ` +
              ex
                .map((x) => {
                  const s = sessionStudent(cls.classObj, sess, x.studentId);
                  return `${s ? s.name : ""}(${TYPE_LABELS[x.type]}:${x.value})`;
                })
                .join("، ") +
              `\n`;
        }
        if (globalFilter.showPosNeg) {
          const pos = sess.scores.filter((x) => x.type === "positive");
          const neg = sess.scores.filter((x) => x.type === "negative");
          if (pos.length)
            text +=
              `   ➕ مثبت (${pos.length}): ` +
              pos
                .map((x) => {
                  const s = sessionStudent(cls.classObj, sess, x.studentId);
                  return `${s ? s.name : ""}(${x.value})`;
                })
                .join("، ") +
              `\n`;
          if (neg.length)
            text +=
              `   ➖ منفی (${neg.length}): ` +
              neg
                .map((x) => {
                  const s = sessionStudent(cls.classObj, sess, x.studentId);
                  return `${s ? s.name : ""}(${x.value})`;
                })
                .join("، ") +
              `\n`;
        }
        if (globalFilter.showHomework) {
          const hw = sess.scores.filter((x) => x.type === "homework");
          if (hw.length)
            text +=
              `   📝 تکالیف (${hw.length}): ` +
              hw
                .map((x) => {
                  const s = sessionStudent(cls.classObj, sess, x.studentId);
                  return `${s ? s.name : ""}${x.note ? "(" + x.note + ")" : ""}`;
                })
                .join("، ") +
              `\n`;
        }
        text += `\n`;
      });
      text += `--------------------------------\n`;
    });
  }
  text += `\n✅ تولید شده توسط کلاس‌یار`;
  return text;
}
